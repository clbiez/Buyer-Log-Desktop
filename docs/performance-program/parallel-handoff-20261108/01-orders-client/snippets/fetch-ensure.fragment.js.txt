function withOrdersFullBeforeChange(reason, stillCurrent, change){
  // Proposed private helper. Preserve the established synchronous full path.
  if(!ordersProjectionIsThin())return change();
  const original=state.orders;
  return ensureOrdersFullForAction(reason).then(()=>{
    if(ordersProjectionIsThin()||state.orders===original||stillCurrent()!==true)
      throw ordersFullActionError('ORDERS_FULL_STALE');
    return change();
  });
}
